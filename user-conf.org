    ;; Helper function to strip quotes from input
    (defun ayrc/strip-quotes (str)
      "Remove leading and trailing quotes from string if present."
      (if (and str (not (string= str "")))
          (let ((stripped (string-trim str)))
            (if (and (>= (length stripped) 2)
                     (or (and (eq (aref stripped 0) ?\")
                              (eq (aref stripped (1- (length stripped))) ?\"))
                         (and (eq (aref stripped 0) ?')
                              (eq (aref stripped (1- (length stripped))) ?'))))
                (substring stripped 1 -1)
              stripped))
        str))

    ;; Doctest helpers: scan current buffer for TEST_CASE/SUBCASE
    (defconst ayrc/adverty--doctest-case-or-subcase-re
      "\\_<\\(TEST_CASE\\|SUBCASE\\)\\_>\\s-*(\\s-*\\(\"\\(?:\\\\.\\|[^\"\\\\]\\)*\"\\)"
      "Regexp matching doctest TEST_CASE/SUBCASE name string literal.")

    (defcustom ayrc/adverty--doctest-scan-time-limit 0.35
      "Max seconds to spend scanning the current buffer for doctest cases."
      :type 'number)

    (defcustom ayrc/adverty--doctest-scan-max-matches 5000
      "Max regexp matches to process when scanning the current buffer."
      :type 'integer)

    (defun ayrc/adverty--doctest--read-c-string (quoted)
      "Decode a C-style QUOTED string literal into an Emacs string.
Falls back to stripping surrounding quotes if decoding fails."
      (condition-case nil
          (read quoted)
        (error (ayrc/strip-quotes quoted))))

    (defun ayrc/adverty--doctest--line-commented-p (pos)
      "Non-nil if POS is after '//' on its line (simple C++ line comment check)."
      (save-excursion
        (goto-char pos)
        (let ((limit pos))
          (goto-char (line-beginning-position))
          (when (search-forward "//" limit t) t))))

    (defun ayrc/adverty--doctest-scan-buffer ()
      "Return an alist of (TEST-CASE . (SUBCASE...)) found in current buffer.

Best-effort: skips matches that appear after '//' on the same line.

If scanning would take too long, returns nil so callers can fall back to manual input."
      (save-excursion
        (goto-char (point-min))
        (let ((current-test nil)
              (tbl (make-hash-table :test 'equal))
              (start (float-time))
              (matches 0)
              (timed-out nil))
          (while (and (not timed-out)
                      (re-search-forward ayrc/adverty--doctest-case-or-subcase-re nil t))
            (setq matches (1+ matches))
            (when (or (> matches ayrc/adverty--doctest-scan-max-matches)
                      (> (- (float-time) start) ayrc/adverty--doctest-scan-time-limit))
              (setq timed-out t))
            (unless timed-out
              (let ((mb (match-beginning 0)))
                (unless (ayrc/adverty--doctest--line-commented-p mb)
                  (let* ((kind (match-string-no-properties 1))
                         (raw (match-string-no-properties 2))
                         (name (ayrc/adverty--doctest--read-c-string raw)))
                    (cond
                     ((string= kind "TEST_CASE")
                      (setq current-test name)
                      (unless (gethash current-test tbl)
                        (puthash current-test '() tbl)))
                     ((string= kind "SUBCASE")
                      (when current-test
                        (let ((xs (gethash current-test tbl)))
                          (unless (member name xs)
                            (puthash current-test (append xs (list name)) tbl)))))))))))
          (unless timed-out
            (let (out)
              (maphash (lambda (k v) (push (cons k v) out)) tbl)
              ;; Stable-ish ordering: alphabetical by test-case, preserve subcase order.
              (setq out (sort out (lambda (a b) (string< (car a) (car b)))))
              out)))))

    (defun ayrc/adverty--doctest-menu-choice ()
      "Prompt user with a menu of TEST_CASE/SUBCASE found in current buffer.
Returns a plist (:test-case STRING-OR-NIL :subcase STRING-OR-NIL)."
      (let* ((cases (ayrc/adverty--doctest-scan-buffer))
             (choices (list (cons "[ALL] run all tests" (list :test-case nil :subcase nil)))))
        (when cases
          (dolist (pair cases)
            (let ((tc (car pair))
                  (subs (cdr pair)))
              (push (cons (format "TEST_CASE: %s" tc)
                          (list :test-case tc :subcase nil))
                    choices)
              (dolist (sc subs)
                (push (cons (format "TEST_CASE: %s  |  SUBCASE: %s" tc sc)
                            (list :test-case tc :subcase sc))
                      choices))))
          (setq choices (nreverse choices)))
        (if (null cases)
            ;; Buffer has no recognizable tests; fall back to manual input.
            (list :test-case (ayrc/strip-quotes
                              (read-string "Enter test case name (or leave empty for all tests): "))
                  :subcase (ayrc/strip-quotes
                            (read-string "Enter subcase name (or leave empty): ")))
          (let* ((picked (completing-read "Select doctest: " choices nil t nil nil (caar choices)))
                 (plist (cdr (assoc picked choices))))
            plist))))

    (defun tsx-comment-line ()
        "Toggle comment on current line or region with JSX support."
        (interactive)
        (if (or (string-equal (file-name-extension buffer-file-name) "tsx")
                (derived-mode-p 'typescript-mode))
                (let* ((start (if (use-region-p) (region-beginning) (line-beginning-position)))
                       (end (if (use-region-p) (region-end) (line-end-position)))
                       (in-jsx (save-excursion
                                   (goto-char start)
                                   (search-forward-regexp "[<>]" end t))))
                    (if in-jsx
                            (progn
                                (save-excursion
                                    ;; Trim leading and trailing whitespace within the region
                                    (goto-char start)
                                    (skip-chars-forward " \t\n")
                                    (setq start (point))
                                    (goto-char end)
                                    (skip-chars-backward " \t\n")
                                    (setq end (point))

                                    ;; Insert JSX comment markers directly around the content
                                    (goto-char end)
                                    (insert "*/}")
                                    (goto-char start)
                                    (insert "{/*")))
                        (comment-line 1)))
            (comment-line 1)))
