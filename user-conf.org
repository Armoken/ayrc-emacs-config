    ;; Helper function to strip quotes from input
    (defun ayrc/strip-quotes (str)
      "Remove leading and trailing quotes from string if present."
      (if (and str (not (string= str "")))
          (let ((stripped (string-trim str)))
            (if (and (>= (length stripped) 2)
                     (or (and (eq (aref stripped 0) ?\")
                              (eq (aref stripped (1- (length stripped))) ?\"))
                         (and (eq (aref stripped 0) ?')
                              (eq (aref stripped (1- (length stripped))) ?'))))
                (substring stripped 1 -1)
              stripped))
        str))

    (defun tsx-comment-line ()
        "Toggle comment on current line or region with JSX support."
        (interactive)
        (if (or (string-equal (file-name-extension buffer-file-name) "tsx")
                (derived-mode-p 'typescript-mode))
                (let* ((start (if (use-region-p) (region-beginning) (line-beginning-position)))
                       (end (if (use-region-p) (region-end) (line-end-position)))
                       (in-jsx (save-excursion
                                   (goto-char start)
                                   (search-forward-regexp "[<>]" end t))))
                    (if in-jsx
                            (progn
                                (save-excursion
                                    ;; Trim leading and trailing whitespace within the region
                                    (goto-char start)
                                    (skip-chars-forward " \t\n")
                                    (setq start (point))
                                    (goto-char end)
                                    (skip-chars-backward " \t\n")
                                    (setq end (point))

                                    ;; Insert JSX comment markers directly around the content
                                    (goto-char end)
                                    (insert "*/}")
                                    (goto-char start)
                                    (insert "{/*")))
                        (comment-line 1)))
            (comment-line 1)))

    (add-hook 'typescript-mode-hook
              (lambda ()
                  (define-key (current-local-map) (kbd "s-/") 'tsx-comment-line)))
